```typescript
import type { Task } from '../../types/task'
import styles from './TaskItem.module.css'

interface TaskItemProps {
  task: Task
  onToggle?: (id: string) => void
  onDelete?: (id: string) => void
}

export function TaskItem({ task, onToggle, onDelete }: TaskItemProps) {
  return (
    <li className={styles.item}>
      <input
        type="checkbox"
        checked={task.completed}
        onChange={() => onToggle?.(task.id)}
        aria-label={`Mark "${task.title}" as ${task.completed ? 'incomplete' : 'complete'}`}
        className={styles.checkbox}
      />
      <span className={task.completed ? styles.titleCompleted : styles.title}>
        {task.title}
      </span>
      <button
        type="button"
        onClick={() => onDelete?.(task.id)}
        aria-label={`Delete "${task.title}"`}
        className={styles.deleteButton}
      >
        Delete
      </button>
    </li>
  )
}
```

Note: Using a text "Delete" label rather than an SVG icon keeps this plan dependency-free and testable by accessible name. Phase 4 or a UI polish pass can replace it with a trash icon without changing the test surface.

**src/App.tsx (full updated file):**
```typescript
import { useState, useEffect, useCallback } from 'react'
import { getTasks, createTask, updateTask, deleteTask } from './api/tasks'
import type { Task } from './types/task'
import { CreateTaskInput } from './components/CreateTaskInput/CreateTaskInput'
import { TaskList } from './components/TaskList/TaskList'
import styles from './App.module.css'

function App() {
  const [tasks, setTasks] = useState<Task[]>([])

  useEffect(() => {
    setTasks(getTasks())
  }, [])

  const handleCreate = useCallback((title: string) => {
    createTask(title)
    setTasks(getTasks())
  }, [])

  const handleToggle = useCallback((id: string) => {
    const task = tasks.find((t) => t.id === id)
    if (!task) return
    updateTask(id, { completed: !task.completed })
    setTasks(getTasks())
  }, [tasks])

  const handleDelete = useCallback((id: string) => {
    deleteTask(id)
    setTasks(getTasks())
  }, [])

  return (
    <main className={styles.container}>
      <h1 className={styles.heading}>TaskTracker</h1>
      <CreateTaskInput onCreate={handleCreate} />
      <TaskList tasks={tasks} onToggle={handleToggle} onDelete={handleDelete} />
    </main>
  )
}

export default App
```

**Design decisions:**
- `handleDelete` calls `deleteTask(id)` then `getTasks()` — same refresh-from-store pattern used throughout Phase 2 and Phase 3
- `useCallback` with `[]` dependency (no closure over `tasks` needed — deleteTask only needs the id)
- If `deleteTask` throws `TASK_NOT_FOUND` (stale ID), the error propagates to the browser console; no toast infrastructure in this plan — same approach as `handleToggle`
- Deleting the last task naturally causes `getTasks()` to return `[]`, which causes TaskList to render the empty state
